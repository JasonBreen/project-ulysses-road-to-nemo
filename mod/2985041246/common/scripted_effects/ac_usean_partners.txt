ac_up_recalculate = {
	if = {
		limit = { ac_up_active = yes }
		if = {
			limit = { tag = SNS }
			set_variable = { ac_up_bonus_surrender_limit = ac_up_sns_continuity }
			multiply_variable = { ac_up_bonus_surrender_limit = 0.0005 }
			set_variable = { ac_up_bonus_stability_factor = ac_up_sns_protection }
			multiply_variable = { ac_up_bonus_stability_factor = 0.0003 }
			set_variable = { ac_up_bonus_supply_consumption_factor = ac_up_sns_dispatch }
			multiply_variable = { ac_up_bonus_supply_consumption_factor = -0.0004 }
			set_variable = { ac_up_programme_burden = 0 }
			add_to_variable = { ac_up_programme_burden = ac_up_sns_continuity }
			add_to_variable = { ac_up_programme_burden = ac_up_sns_protection }
			add_to_variable = { ac_up_programme_burden = ac_up_sns_dispatch }
			multiply_variable = { ac_up_programme_burden = 0.00015 }
			set_variable = { ac_up_public_record_bonus = 0 }
			set_variable = { ac_up_readiness_record_bonus = 0 }
			if = {
				limit = { has_country_flag = ac_up_public_record }
				set_variable = { ac_up_public_record_bonus = 0.005 }
				add_to_variable = { ac_up_programme_burden = 0.003 }
			}
			if = {
				limit = { has_country_flag = ac_up_readiness_record }
				set_variable = { ac_up_readiness_record_bonus = 0.01 }
				add_to_variable = { ac_up_programme_burden = 0.005 }
			}
			add_to_variable = { ac_up_bonus_stability_factor = ac_up_public_record_bonus }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ac_up_sns_programme } } }
				add_dynamic_modifier = { modifier = ac_up_sns_programme }
			}
		}
		if = {
			limit = { tag = AMB }
			set_variable = { ac_up_bonus_industrial_capacity_factory = ac_up_amb_development }
			multiply_variable = { ac_up_bonus_industrial_capacity_factory = 0.00025 }
			set_variable = { ac_up_bonus_supply_consumption_factor = ac_up_amb_service }
			multiply_variable = { ac_up_bonus_supply_consumption_factor = -0.0003 }
			set_variable = { ac_up_bonus_production_speed_buildings_factor = ac_up_amb_transport }
			multiply_variable = { ac_up_bonus_production_speed_buildings_factor = 0.00035 }
			set_variable = { ac_up_programme_burden = 0 }
			add_to_variable = { ac_up_programme_burden = ac_up_amb_development }
			add_to_variable = { ac_up_programme_burden = ac_up_amb_service }
			add_to_variable = { ac_up_programme_burden = ac_up_amb_transport }
			multiply_variable = { ac_up_programme_burden = 0.00015 }
			set_variable = { ac_up_public_record_bonus = 0 }
			set_variable = { ac_up_readiness_record_bonus = 0 }
			if = {
				limit = { has_country_flag = ac_up_public_record }
				set_variable = { ac_up_public_record_bonus = 0.005 }
				add_to_variable = { ac_up_programme_burden = 0.003 }
			}
			if = {
				limit = { has_country_flag = ac_up_readiness_record }
				set_variable = { ac_up_readiness_record_bonus = 0.01 }
				add_to_variable = { ac_up_programme_burden = 0.005 }
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ac_up_amb_programme } } }
				add_dynamic_modifier = { modifier = ac_up_amb_programme }
			}
			set_variable = { ac_up_amber_limit = 2 }
			add_to_variable = { ac_up_amber_limit = ac_up_amb_service_level }
		}
		if = {
			limit = { tag = UGE }
			set_variable = { ac_up_bonus_supply_consumption_factor = ac_up_uge_dispersal }
			multiply_variable = { ac_up_bonus_supply_consumption_factor = -0.0004 }
			set_variable = { ac_up_bonus_army_speed_factor = ac_up_uge_mobility }
			multiply_variable = { ac_up_bonus_army_speed_factor = 0.00035 }
			set_variable = { ac_up_bonus_surrender_limit = ac_up_uge_reserve }
			multiply_variable = { ac_up_bonus_surrender_limit = 0.00025 }
			set_variable = { ac_up_programme_burden = 0 }
			add_to_variable = { ac_up_programme_burden = ac_up_uge_dispersal }
			add_to_variable = { ac_up_programme_burden = ac_up_uge_mobility }
			add_to_variable = { ac_up_programme_burden = ac_up_uge_reserve }
			multiply_variable = { ac_up_programme_burden = 0.00015 }
			set_variable = { ac_up_public_record_bonus = 0 }
			set_variable = { ac_up_readiness_record_bonus = 0 }
			if = {
				limit = { has_country_flag = ac_up_public_record }
				set_variable = { ac_up_public_record_bonus = 0.005 }
				add_to_variable = { ac_up_programme_burden = 0.003 }
			}
			if = {
				limit = { has_country_flag = ac_up_readiness_record }
				set_variable = { ac_up_readiness_record_bonus = 0.01 }
				add_to_variable = { ac_up_programme_burden = 0.005 }
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ac_up_uge_programme } } }
				add_dynamic_modifier = { modifier = ac_up_uge_programme }
			}
			set_variable = { ac_up_uge_limit = 2 }
			if = {
				limit = { check_variable = { ac_up_uge_reserve_level > 1 } }
				add_to_variable = { ac_up_uge_limit = 1 }
			}
			if = {
				limit = { check_variable = { ac_up_uge_reserve_level > 3 } }
				add_to_variable = { ac_up_uge_limit = 1 }
			}
		}
		if = {
			limit = { tag = DEL }
			set_variable = { ac_up_bonus_stability_factor = ac_up_del_oversight }
			multiply_variable = { ac_up_bonus_stability_factor = 0.0003 }
			set_variable = { ac_up_bonus_planning_speed = ac_up_del_command }
			multiply_variable = { ac_up_bonus_planning_speed = 0.0004 }
			set_variable = { ac_up_bonus_research_speed_factor = ac_up_del_records }
			multiply_variable = { ac_up_bonus_research_speed_factor = 0.0002 }
			set_variable = { ac_up_programme_burden = 0 }
			add_to_variable = { ac_up_programme_burden = ac_up_del_oversight }
			add_to_variable = { ac_up_programme_burden = ac_up_del_command }
			add_to_variable = { ac_up_programme_burden = ac_up_del_records }
			multiply_variable = { ac_up_programme_burden = 0.00015 }
			set_variable = { ac_up_public_record_bonus = 0 }
			set_variable = { ac_up_readiness_record_bonus = 0 }
			if = {
				limit = { has_country_flag = ac_up_public_record }
				set_variable = { ac_up_public_record_bonus = 0.005 }
				add_to_variable = { ac_up_programme_burden = 0.003 }
			}
			if = {
				limit = { has_country_flag = ac_up_readiness_record }
				set_variable = { ac_up_readiness_record_bonus = 0.01 }
				add_to_variable = { ac_up_programme_burden = 0.005 }
			}
			set_temp_variable = { ac_up_unclosed_dockets = ac_up_del_obligations }
			multiply_temp_variable = { ac_up_unclosed_dockets = 0.002 }
			add_to_variable = { ac_up_programme_burden = ac_up_unclosed_dockets }
			add_to_variable = { ac_up_bonus_stability_factor = ac_up_public_record_bonus }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ac_up_del_programme } } }
				add_dynamic_modifier = { modifier = ac_up_del_programme }
			}
		}
	}
}

	}
