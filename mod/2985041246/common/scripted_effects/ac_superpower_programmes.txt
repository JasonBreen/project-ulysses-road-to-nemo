ac_snp_sync_standing = {
	if = {
		limit = {
			ac_snp_participant = yes
		}
		if = {
			limit = {
				ac_snp_active = yes tag = OSE original_tag = OSE check_variable = { ac_snp_ose_public >= 65 }
			}
			add_ideas = ac_snp_ose_public_standing
		}
		else = {
			remove_ideas = ac_snp_ose_public_standing
		}
		if = {
			limit = {
				ac_snp_active = yes tag = OSE original_tag = OSE check_variable = { ac_snp_ose_security >= 65 }
			}
			add_ideas = ac_snp_ose_security_standing
		}
		else = {
			remove_ideas = ac_snp_ose_security_standing
		}
		if = {
			limit = {
				ac_snp_active = yes tag = YUK original_tag = YUK check_variable = { ac_snp_yuk_civil >= 60 }
			}
			add_ideas = ac_snp_yuk_civil_standing
		}
		else = {
			remove_ideas = ac_snp_yuk_civil_standing
		}
		if = {
			limit = {
				ac_snp_active = yes tag = YUK original_tag = YUK check_variable = { ac_snp_yuk_command >= 60 }
			}
			add_ideas = ac_snp_yuk_command_standing
		}
		else = {
			remove_ideas = ac_snp_yuk_command_standing
		}
	}
}

OSE_snp_appropriations_service_1_start = {
	clr_country_flag = ac_snp_payment_authorized
	if = {
		limit = {
			OSE_snp_appropriations_service_1_can_start = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_snp_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_snp_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_snp_payment_authorized
			}
			clr_country_flag = ac_snp_payment_authorized
			if = {
				limit = {
					check_variable = { ac_snp_ose_public >= 65 }
				}
				add_political_power = -30
			}
			else = {
				add_political_power = -35
			}
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
			set_country_flag = OSE_snp_appropriations_service_1_active
			set_variable = { ac_snp_ose_appropriations_1_started = date }
			set_variable = { ac_snp_ose_appropriations_1_due = date }
			add_to_variable = { ac_snp_ose_appropriations_1_due = 90 }
			set_variable = { ac_snp_ose_appropriations_1_elapsed = 0 }
			if = {
				limit = {
					has_war = yes
				}
				set_country_flag = OSE_snp_appropriations_service_1_wartime
				add_timed_idea = { idea = OSE_snp_appropriations_service_1_domestic_spirit days = 90 }
			}
			else = {
				clr_country_flag = OSE_snp_appropriations_service_1_wartime
				add_timed_idea = { idea = OSE_snp_appropriations_service_1_spirit days = 90 }
			}
			country_event = { id = ac_snp.1000 days = 90 }
			ac_snp_reconcile = yes
		}
	}
}

OSE_snp_appropriations_service_1_finish = {
	OSE_snp_appropriations_service_1_progress = yes
	if = {
		limit = {
			OSE_snp_appropriations_service_1_finishable = yes check_variable = { ac_snp_ose_appropriations_1_elapsed >= 90 }
		}
		set_country_flag = OSE_snp_appropriations_service_1_completed
		set_country_flag = OSE_snp_appropriations_service_completed
		add_to_variable = { ac_snp_completed = 1 }
		set_variable = { ac_snp_ose_appropriations_1_elapsed = 90 }
		remove_ideas = OSE_snp_appropriations_service_1_spirit
		remove_ideas = OSE_snp_appropriations_service_1_domestic_spirit
		clr_country_flag = OSE_snp_appropriations_service_1_active
		if = {
			limit = {
				has_country_flag = OSE_snp_appropriations_service_1_wartime
			}
			set_country_flag = OSE_snp_appropriations_last_domestic
		}
		else = {
			clr_country_flag = OSE_snp_appropriations_last_domestic
		}
		clr_country_flag = OSE_snp_appropriations_service_1_wartime
		set_country_flag = { flag = OSE_snp_appropriations_service_1_cooldown days = 90 }
		add_to_variable = { ac_snp_ose_public = 3 }
		add_to_variable = { ac_snp_ose_security = 1 }
		ac_snp_reconcile = yes
	}
}
