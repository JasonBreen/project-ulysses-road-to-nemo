ac_econ_enabled = {
	has_game_rule = { rule = ac_economy option = experimental }
	any_owned_state = { always = yes }
}

ac_econ_ready = {
	ac_econ_enabled = yes
	has_country_flag = ac_econ_initialized
}

ac_econ_can_borrow = {
	ac_econ_ready = yes
	check_variable = { ac_econ_arrears = 0 }
	check_variable = { var = ac_econ_credit_room value = 5 compare = greater_than_or_equals }
}

ac_econ_can_repay = {
	ac_econ_ready = yes
	check_variable = { ac_econ_arrears = 0 }
	check_variable = { ac_econ_debt > 0 }
	check_variable = { ac_econ_treasury > 0 }
}

ac_econ_can_produce = {
	ac_econ_ready = yes
	has_capitulated = no
	has_tech = infantry_weapons
	num_of_military_factories > 0
}

ac_econ_can_order = {
	ac_econ_can_produce = yes
	check_variable = { ac_econ_arrears = 0 }
	check_variable = { ac_econ_procurement_weeks = 0 }
	check_variable = { var = ac_econ_treasury value = 5 compare = greater_than_or_equals }
}
