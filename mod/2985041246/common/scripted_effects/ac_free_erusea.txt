ac_fe_initialize = {
	if = {
		limit = {
			ac_fe_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_fe_current_service }
			}
			set_variable = { ac_fe_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_fe_elapsed }
			}
			set_variable = { ac_fe_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_fe_completed_services }
			}
			set_variable = { ac_fe_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_fe_last_delivered_days }
			}
			set_variable = { ac_fe_last_delivered_days = 0 }
		}
		set_country_flag = ac_fe_initialized
	}
}

ac_fe_pay_programme = {
	clr_country_flag = ac_fe_payment_authorized
	if = {
		limit = {
			ac_fe_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_fe_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_fe_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_fe_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_fe_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_fe_service_active has_variable = ac_fe_service_started
		}
		set_variable = { ac_fe_elapsed = global.date }
		subtract_from_variable = { ac_fe_elapsed = ac_fe_service_started }
		clamp_variable = { var = ac_fe_elapsed min = 0 max = 90 }
	}
}

ac_fe_update_tier = {
	if = {
		limit = {
			ac_fe_running = yes check_variable = { ac_fe_pressure > 69.99 }
		}
		if = {
			limit = {
				NOT = { has_idea = ac_fe_pressure_severe }
			}
			remove_ideas = ac_fe_pressure_severe
			remove_ideas = ac_fe_pressure_active
			remove_ideas = ac_fe_pressure_low
			add_ideas = ac_fe_pressure_severe
		}
	}
	else_if = {
		limit = {
			ac_fe_running = yes check_variable = { ac_fe_pressure > 34.99 }
		}
		if = {
			limit = {
				NOT = { has_idea = ac_fe_pressure_active }
			}
			remove_ideas = ac_fe_pressure_severe
			remove_ideas = ac_fe_pressure_active
			remove_ideas = ac_fe_pressure_low
			add_ideas = ac_fe_pressure_active
		}
	}
	else_if = {
		limit = {
			ac_fe_running = yes
		}
		if = {
			limit = {
				NOT = { has_idea = ac_fe_pressure_low }
			}
			remove_ideas = ac_fe_pressure_severe
			remove_ideas = ac_fe_pressure_active
			remove_ideas = ac_fe_pressure_low

		}
	}
	else = {

		remove_ideas = ac_fe_pressure_low
	}
}

ac_fe_drift = {
	set_temp_variable = { ac_fe_delta = 0.4 }
	if = {
		limit = {
			OR = { has_idea = ERU_fe_insurrection_policy_a_spirit has_idea = ERU_fe_insurrection_policy_a_spirit_settled }
		}
		subtract_from_temp_variable = { ac_fe_delta = 0.7 }
	}
	if = {
		limit = {
			OR = { has_idea = ERU_fe_insurrection_policy_b_spirit has_idea = ERU_fe_insurrection_policy_b_spirit_settled }
		}
		subtract_from_temp_variable = { ac_fe_delta = 0.9 }
	}
	if = {
		limit = {
			has_idea = ac_fe_emergency_administration
		}
		subtract_from_temp_variable = { ac_fe_delta = 0.5 }
	}
	if = {
		limit = {
			has_idea = ac_fe_partner_assistance
		}
		subtract_from_temp_variable = { ac_fe_delta = 0.5 }
	}
	if = {
		limit = {
			has_stability < 0.35
		}
		add_to_temp_variable = { ac_fe_delta = 0.3 }
	}
	add_to_variable = { ac_fe_pressure = ac_fe_delta }
	clamp_variable = { var = ac_fe_pressure min = 0 max = 100 }
}

ac_fe_owner_weekly = {
	if = {
		limit = {
			ac_fe_owner_eligible = yes NOT = { has_global_flag = ac_fe_insurrection_started } NOT = { has_global_flag = ac_fe_insurrection_closed } date > 2006.9.25
		}
		if = {
			limit = {
				date < 2007.9.26
			}
			ac_fe_begin = yes
		}
		else = {
			set_global_flag = ac_fe_insurrection_closed
			set_global_flag = ac_fe_late_start_skipped
		}
	}
	if = {
		limit = {
			ac_fe_running = yes
		}
		if = {
			limit = {
				ac_fe_owner_eligible = no
			}
			ac_fe_close = yes
		}
		else = {
			if = {
				limit = {
					NOT = { has_global_flag = ac_fe_prolonged } NOT = { has_country_flag = ac_fe_response_window }
				}
				set_global_flag = ac_fe_prolonged
			}
			ac_fe_drift = yes
			ac_fe_update_tier = yes
			if = {
				limit = {
					check_variable = { ac_fe_pressure < 5 }
				}
				ac_fe_contain = yes
			}
			else_if = {
				limit = {
					check_variable = { ac_fe_pressure > 84.99 } NOT = { has_country_flag = ac_fe_crisis_cooldown }
				}
				set_country_flag = { flag = ac_fe_crisis_cooldown days = 365 }
				country_event = { id = ac_fe.2 days = 1 }
			}
		}
	}
}
