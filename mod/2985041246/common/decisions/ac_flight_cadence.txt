# Repeatable static fires, then a pad. Provisional gameplay.
# The Ulysses clock is the 8 July 1999 planetfall already used by Arkbird.
ac_flight_cadence_cat = {
	ac_flight_static_fire = {
		icon = GFX_decision_ac_sp_arkbird
		allowed = {
			OR = {
				original_tag = OSE
				original_tag = YUK
				original_tag = ERU
			}
		}
		visible = {
			date < 2006.1.1
		}
		available = {
			has_political_power > 19
		}
		cost = 20
		days_remove = 14
		ai_will_do = {
			base = 40
			modifier = {
				factor = 0
				has_idea = ac_flight_cadence
			}
			modifier = {
				factor = 0
				check_variable = { ac_flight_success > 5 }
			}
		}
		complete_effect = {
			if = {
				limit = { NOT = { has_country_flag = ac_flight_program_open } }
				set_country_flag = ac_flight_program_open
				set_variable = { ac_flight_success = 0 }
			}
		}
		remove_effect = {
			ac_flight_static_fire_effect = yes
		}
	}

	ac_flight_claim_cadence = {
		icon = GFX_decision_ac_sp_arkbird
		allowed = {
			OR = {
				original_tag = OSE
				original_tag = YUK
				original_tag = ERU
			}
		}
		visible = {
			has_country_flag = ac_flight_program_open
			NOT = { has_idea = ac_flight_cadence }
		}
		available = {
			check_variable = { ac_flight_success > 2 }
		}
		fire_only_once = yes
		cost = 25
		days_remove = 7
		ai_will_do = {
			base = 100
		}
		remove_effect = {
			add_ideas = ac_flight_cadence
		}
	}

	ac_flight_raise_a_pad = {
		icon = GFX_decision_ac_sp_arkbird
		allowed = {
			OR = {
				original_tag = OSE
				original_tag = YUK
				original_tag = ERU
			}
		}
		visible = {
			has_idea = ac_flight_cadence
			NOT = { has_country_flag = ac_flight_pad_raised }
		}
		available = {
			has_idea = ac_flight_cadence
		}
		fire_only_once = yes
		cost = 50
		days_remove = 70
		ai_will_do = {
			base = 50
		}
		remove_effect = {
			set_country_flag = ac_flight_pad_raised
			if = {
				limit = { NOT = { has_tech = experimental_rockets } }
				set_technology = { experimental_rockets = 1 popup = no }
			}
			capital_scope = {
				add_building_construction = {
					type = rocket_site
					level = 1
					instant_build = yes
				}
			}
		}
	}

	ac_flight_ulysses_clock = {
		icon = GFX_decision_ac_sp_arkbird
		allowed = {
			OR = {
				original_tag = OSE
				original_tag = YUK
				original_tag = ERU
			}
		}
		activation = {
			always = no
		}
		available = {
			always = no
		}
		selectable_mission = no
		days_mission_timeout = 4000
		cancel_trigger = {
			OR = {
				has_country_flag = ac_flight_planetfall_resolved
				date > 1999.7.8
			}
		}
		timeout_effect = {
			if = {
				limit = {
					NOT = { has_country_flag = ac_flight_planetfall_resolved }
					date > 1999.7.7
					NOT = { date > 1999.7.20 }
				}
				set_country_flag = ac_flight_planetfall_resolved
				country_event = ac_flight_cadence.3
			}
		}
	}
}
