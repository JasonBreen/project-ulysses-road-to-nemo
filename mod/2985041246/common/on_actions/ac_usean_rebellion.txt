on_actions = {
	on_weekly_FCU = {
		effect = {
			if = {
				limit = {
					date > 1995.12.30
					NOT = { has_country_flag = ac_ur_skully_recorded }
				}
				set_country_flag = ac_ur_skully_recorded
				country_event = { id = ac_ur.1 days = 1 }
			}
			if = {
				limit = {
					date > 1997.5.29
					NOT = { has_country_flag = ac_ur_coup_started }
				}
				set_country_flag = ac_ur_coup_started
				add_timed_idea = { idea = ac_ur_coup_strain days = 120 }
				country_event = { id = ac_ur.2 days = 1 }
				every_country = {
					limit = {
						exists = yes
						NOT = { original_tag = FCU }
						OR = {
							is_in_faction_with = FCU
							original_tag = ERU
						}
					}
					country_event = { id = ac_ur.4 days = 1 }
				}
			}
		}
	}
}
