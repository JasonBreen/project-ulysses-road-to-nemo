ac_alc_initialize = {
	if = {
		limit = {
			ac_alc_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_alc_current_service }
			}
			set_variable = { ac_alc_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_alc_elapsed }
			}
			set_variable = { ac_alc_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_alc_completed_services }
			}
			set_variable = { ac_alc_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_alc_last_delivered_days }
			}
			set_variable = { ac_alc_last_delivered_days = 0 }
		}
		set_country_flag = ac_alc_initialized
	}
}

ac_alc_pay_programme = {
	clr_country_flag = ac_alc_payment_authorized
	if = {
		limit = {
			ac_alc_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_alc_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_alc_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_alc_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_alc_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_alc_service_active has_variable = ac_alc_service_started
		}
		set_variable = { ac_alc_elapsed = date }
		subtract_from_variable = { ac_alc_elapsed = ac_alc_service_started }
		clamp_variable = { var = ac_alc_elapsed min = 0 max = 90 }
	}
}

ac_alc_calendar_tick = {
	if = {
		limit = {
			date > 2020.8.31 NOT = { has_global_flag = ac_alc_calendar_started } NOT = { has_global_flag = ac_alc_closed }
		}
		if = {
			limit = {
				date < 2020.11.1 ac_alc_principals = yes has_global_flag = ac_lcw_ended
			}
			set_global_flag = ac_alc_calendar_started
		}
		else_if = {
			limit = {
				date < 2020.11.1 NOT = { has_global_flag = ac_lcw_ended }
			}
			set_global_flag = ac_alc_closed
			set_global_flag = ac_alc_route_unreachable
		}
		else_if = {
			limit = {
				NOT = { ac_alc_principals = yes }
			}
			set_global_flag = ac_alc_closed
			set_global_flag = ac_alc_principal_missing
		}
		else = {
			set_global_flag = ac_alc_closed
			set_global_flag = ac_alc_late_start_skipped
		}
	}
}
