# Project Ulysses generic prototype rewards (provisional gameplay).
# ROOT is the project, FROM is the country inside weight blocks; inside
# iteration_output.country_effects FROM is the project.

ac_sp_reward_breakthrough = {
	icon = GFX_ac_sp_reward_breakthrough
	fire_only_once = yes
	threshold = {
		min = 0
		max = 100
	}
	weight = {
		base = 10
	}
	option = {
		token = ac_sp_reward_breakthrough_option
		iteration_output = {
			country_effects = {
				FROM = { add_project_progress_ratio = constant:sp_progress.gain.medium }
			}
		}
	}
}

ac_sp_reward_oversight_review = {
	icon = GFX_ac_sp_reward_oversight_review
	fire_only_once = yes
	threshold = {
		min = 0
		max = 100
	}
	weight = {
		base = 6
	}
	option = {
		token = ac_sp_reward_oversight_review_accelerate
		default = yes
		iteration_output = {
			country_effects = {
				add_political_power = -25
				FROM = { add_project_progress_ratio = constant:sp_progress.gain.low }
			}
		}
	}
	option = {
		token = ac_sp_reward_oversight_review_audit
		iteration_output = {
			country_effects = {
				add_political_power = 25
			}
		}
	}
}

ac_sp_reward_field_trials_orbital = {
	icon = GFX_ac_sp_reward_field_trials_orbital
	fire_only_once = yes
	threshold = {
		min = 0
		max = 100
	}
	weight = {
		base = 8
		modifier = {
			factor = 0
			var:scientist = {

		}

			scientist_effects = {

		}
	}
}
