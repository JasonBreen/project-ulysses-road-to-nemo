ac_lcw_initialize = {
	if = {
		limit = {
			ac_lcw_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_lcw_current_service }
			}
			set_variable = { ac_lcw_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_lcw_elapsed }
			}
			set_variable = { ac_lcw_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_lcw_completed_services }
			}
			set_variable = { ac_lcw_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_lcw_last_delivered_days }
			}
			set_variable = { ac_lcw_last_delivered_days = 0 }
		}
		set_country_flag = ac_lcw_initialized
	}
}

ac_lcw_pay_programme = {
	clr_country_flag = ac_lcw_payment_authorized
	if = {
		limit = {
			ac_lcw_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_lcw_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_lcw_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_lcw_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_lcw_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_lcw_service_active has_variable = ac_lcw_service_started
		}
		set_variable = { ac_lcw_elapsed = date }
		subtract_from_variable = { ac_lcw_elapsed = ac_lcw_service_started }
		clamp_variable = { var = ac_lcw_elapsed min = 0 max = 90 }
	}
}

ac_lcw_owner_weekly = {
	if = {
		limit = {
			tag = LEA original_tag = LEA NOT = { has_global_flag = ac_lcw_closed }
		}
		if = {
			limit = {
				date > 2008.12.31 NOT = { has_global_flag = ac_lcw_started } NOT = { has_global_flag = ac_lcw_late_start_skipped }
			}
			if = {
				limit = {
					date < 2010.1.1 has_capitulated = no
				}
				set_global_flag = ac_lcw_started
				set_variable = { ac_lcw_authority = 40 }
				add_ideas = ac_lcw_civil_war
				news_event = { id = ac_lcw.1 days = 1 }
			}
			else = {
				set_global_flag = ac_lcw_late_start_skipped
				set_global_flag = ac_lcw_closed
			}
		}
		if = {
			limit = {
				ac_lcw_running = yes
			}
			if = {
				limit = {
					date > 2018.12.31 date < 2020.1.1 check_variable = { ac_lcw_authority > 49 }
				}
				set_global_flag = ac_lcw_ended
				set_global_flag = ac_lcw_historical_end
				remove_ideas = ac_lcw_civil_war
				add_timed_idea = { idea = ac_lcw_war_over days = 365 }
				news_event = { id = ac_lcw.2 days = 1 }
			}
			else_if = {
				limit = {
					NOT = { date < 2020.1.1 } NOT = { has_global_flag = ac_lcw_prolonged }
				}
				set_global_flag = ac_lcw_prolonged
				news_event = { id = ac_lcw.3 days = 1 }
			}
			else_if = {
				limit = {
					has_global_flag = ac_lcw_prolonged check_variable = { ac_lcw_authority > 49 }
				}
				set_global_flag = ac_lcw_ended
				set_global_flag = ac_lcw_late_end
				remove_ideas = ac_lcw_civil_war
				add_timed_idea = { idea = ac_lcw_war_over days = 365 }
				news_event = { id = ac_lcw.4 days = 1 }
			}
		}
	}
}
