	is_triggered_only = yes

	option = {
		name = ac_fe.2.a
		ai_chance = { factor = 60 }
		set_country_flag = ac_fe_assistance_requested

		name = ac_fe.2.b
		ai_chance = { factor = 40 }

		ac_fe_relief_service = yes
	}
}
