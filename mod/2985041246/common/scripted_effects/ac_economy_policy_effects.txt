ac_econ_set_low_tax = {
	if = {
		limit = {
			ac_econ_ready = yes
			NOT = { has_country_flag = ac_econ_tax_cooldown }
			NOT = { check_variable = { ac_econ_tax_level = 0 } }
		}
		set_variable = { ac_econ_tax_level = 0 }
		set_country_flag = { flag = ac_econ_tax_cooldown days = 30 }
		ac_econ_refresh_budget = yes
	}
}
