# Original provisional gameplay; source: tools/content_waves (The Oversight Years). Native acceptance pending.
on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = ERU } ERU = { ac_fe_initialize = yes } }
			if = { limit = { country_exists = FCU } FCU = { ac_fe_initialize = yes } }
		}
	}
	on_weekly_ERU = {
		effect = {
			ac_fe_weekly_eru = yes
		}
	}
	on_weekly_FCU = {
		effect = {
			ac_fe_weekly_fcu = yes
		}
	}
	on_annex = {
		effect = {
			FROM = { ac_fe_interrupt = yes }
		}
	}
	on_capitulation_immediate = {
		effect = {
			ac_fe_interrupt = yes
		}
	}
	on_subject_annexed = {
		effect = {
			ac_fe_interrupt = yes
		}
	}
	on_war_relation_added = {
		effect = {
			ac_fe_interrupt_service = yes
			FROM = { ac_fe_interrupt_service = yes }
		}
	}
}
