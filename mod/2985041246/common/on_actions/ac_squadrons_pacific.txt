on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					ac_sqp_host = yes
				}
				ac_sqp_tick = yes
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					ac_sqp_host = yes
				}
				ac_sqp_tick = yes
			}
		}
	}
on_annex = {
	effect = {
		FROM = {
			ac_sqp_interrupt = yes
		}
	}
}
on_subject_annexed = {
	effect = {
		ac_sqp_interrupt = yes
	}
}
}
