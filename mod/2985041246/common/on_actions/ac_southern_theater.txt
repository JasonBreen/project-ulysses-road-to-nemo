# Original provisional gameplay; source: tools/content_waves (The Aurelia-Leasath War). Native acceptance pending.
on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = AUR } AUR = { ac_alc_initialize = yes } }
			if = { limit = { country_exists = LEA } LEA = { ac_alc_initialize = yes } }
			if = { limit = { country_exists = LEA } LEA = { ac_alc_restore = yes } }
		}
	}
	on_weekly_AUR = {
		effect = {
			ac_alc_weekly_aur = yes
			ac_alc_owner_weekly_aur = yes
		}
	}
	on_weekly_LEA = {
		effect = {
			ac_alc_weekly_lea = yes
		}
	}
	on_annex = {
		effect = {
			FROM = { ac_alc_interrupt = yes }
			if = { limit = { OR = { tag = AUR tag = LEA } } FROM = { ac_alc_interrupt_principal = yes } }
			else = { FROM = { ac_alc_interrupt_external = yes } }
		}
	}
	on_capitulation_immediate = {
		effect = {
			ac_alc_interrupt = yes
			if = { limit = { FROM = { OR = { tag = AUR tag = LEA } } } ac_alc_interrupt_principal = yes }
			else = { ac_alc_interrupt_external = yes }
		}
	}
	on_subject_annexed = {
		effect = {
			ac_alc_interrupt = yes
			if = { limit = { FROM = { OR = { tag = AUR tag = LEA } } } ac_alc_interrupt_principal = yes }
			else = { ac_alc_interrupt_external = yes }
		}
	}
	on_war_relation_added = {
		effect = {
			ac_alc_interrupt_service = yes
			FROM = { ac_alc_interrupt_service = yes }
		}
	}
}
