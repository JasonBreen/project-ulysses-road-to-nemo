ac_rc_initialize = {
	if = {
		limit = {
			ac_rc_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_rc_current_service }
			}
			set_variable = { ac_rc_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_rc_elapsed }
			}
			set_variable = { ac_rc_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_rc_completed_services }
			}
			set_variable = { ac_rc_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_rc_last_delivered_days }
			}
			set_variable = { ac_rc_last_delivered_days = 0 }
		}
		set_country_flag = ac_rc_initialized
	}
}

ac_rc_pay_programme = {
	clr_country_flag = ac_rc_payment_authorized
	if = {
		limit = {
			ac_rc_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_rc_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_rc_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_rc_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_rc_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_rc_service_active has_variable = ac_rc_service_started
		}
		set_variable = { ac_rc_elapsed = date }
		subtract_from_variable = { ac_rc_elapsed = ac_rc_service_started }
		clamp_variable = { var = ac_rc_elapsed min = 0 max = 90 }
	}
}
