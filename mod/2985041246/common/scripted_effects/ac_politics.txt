ac_pol_refresh_form = {
	clr_country_flag = ac_pol_form_civil_administration
	clr_country_flag = ac_pol_form_executive_office
	clr_country_flag = ac_pol_form_crown_office
	clr_country_flag = ac_pol_form_party_secretariat
	clr_country_flag = ac_pol_form_command_administration
	clr_country_flag = ac_pol_form_council_office
	if = {
		limit = {
			OR = { original_tag = AWW original_tag = RAT }
		}
		set_country_flag = ac_pol_form_council_office
	}
	else = {
		if = {
			limit = {
				has_government = democratic
			}
			set_country_flag = ac_pol_form_civil_administration
		}
		else_if = {
			limit = {
				has_government = monarchism
			}
			set_country_flag = ac_pol_form_crown_office
		}
		else_if = {
			limit = {
				has_government = communism
			}
			set_country_flag = ac_pol_form_party_secretariat
		}
		else_if = {
			limit = {
				has_government = fascism
			}
			set_country_flag = ac_pol_form_command_administration
		}
		else = {
			set_country_flag = ac_pol_form_executive_office
		}
	}
}

ac_pol_bounds = {
	clamp_variable = { var = ac_pol_capacity min = 0 max = 100 }
}

ac_pol_change_capacity = {
	if = {
		limit = {
			ac_pol_active = yes
		}
		add_to_variable = { ac_pol_capacity = ac_pol_capacity_delta }
		ac_pol_bounds = yes
		if = {
			limit = {
				check_variable = { ac_pol_capacity < 1 }
			}
			ac_pol_suspend_policy = yes
		}
	}
}

ac_pol_sync_fiscal = {
	if = {
		limit = {
			ac_macro_ready = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_pol_fiscal_identity }
			}
			set_variable = { ac_pol_fiscal_applied = 0 }
			set_variable = { ac_pol_fiscal_identity = THIS.id }
			set_country_flag = ac_pol_fiscal_dirty
		}
		if = {
			limit = {
				NOT = { check_variable = { var = ac_pol_fiscal_identity value = THIS.id compare = equals } }
			}
			set_variable = { ac_pol_fiscal_applied = 0 }
			set_variable = { ac_pol_fiscal_identity = THIS.id }
			set_country_flag = ac_pol_fiscal_dirty
		}
		if = {
			limit = {
				has_country_flag = ac_pol_fiscal_dirty
			}
			set_temp_variable = { ac_macro_commitment_delta = ac_pol_fiscal_applied }
			multiply_temp_variable = { ac_macro_commitment_delta = -1 }
			ac_macro_change_commitment = yes
			set_variable = { ac_pol_fiscal_applied = 0 }
			if = {
				limit = {
					has_country_flag = ac_pol_policy_civic
				}
				set_variable = { ac_pol_fiscal_applied = ac_macro_nominal_gdp }
				multiply_variable = { ac_pol_fiscal_applied = 0.0005 }
			}
			if = {
				limit = {
					has_country_flag = ac_pol_policy_military
				}
				set_variable = { ac_pol_fiscal_applied = ac_macro_nominal_gdp }
				multiply_variable = { ac_pol_fiscal_applied = 0.001 }
			}
			if = {
				limit = {
					has_country_flag = ac_pol_policy_district
				}
				set_variable = { ac_pol_fiscal_applied = ac_macro_nominal_gdp }
				multiply_variable = { ac_pol_fiscal_applied = 0.0003 }
			}
			set_temp_variable = { ac_macro_commitment_delta = ac_pol_fiscal_applied }
			ac_macro_change_commitment = yes
			clr_country_flag = ac_pol_fiscal_dirty
		}
	}
}

ac_pol_clear_policy = {
	remove_ideas = ac_pol_civic
	clr_country_flag = ac_pol_policy_civic
	remove_ideas = ac_pol_military
	clr_country_flag = ac_pol_policy_military
	remove_ideas = ac_pol_district
	clr_country_flag = ac_pol_policy_district
	set_country_flag = ac_pol_fiscal_dirty
}

	if = {
		limit = {
			ac_pol_can_adopt = yes NOT = { has_country_flag = ac_pol_policy_civic }
		}

ac_pol_weekly = {
	ac_pol_initialize = yes
	ac_pol_repair_reviews = yes
	if = {
		limit = {
			ac_pol_active = yes
		}
		ac_pol_refresh_form = yes
		ac_pol_sync_fiscal = yes
		if = {
			limit = {
				NOT = { has_country_flag = ac_pol_weekly_paid }
			}
			set_country_flag = { flag = ac_pol_weekly_paid days = 6 }
			if = {
				limit = {
					has_country_flag = ac_pol_policy_civic
				}
				add_to_variable = { ac_pol_capacity = -1 }
			}
			if = {
				limit = {
					has_country_flag = ac_pol_policy_military
				}
				add_to_variable = { ac_pol_capacity = -2 }
			}
			if = {
				limit = {
					has_country_flag = ac_pol_policy_district
				}
				add_to_variable = { ac_pol_capacity = -1 }
			}
			ac_pol_bounds = yes
			if = {
				limit = {
					check_variable = { ac_pol_capacity < 1 }
				}
				ac_pol_suspend_policy = yes
			}
		}
	}
}
