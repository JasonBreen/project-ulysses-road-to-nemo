ac_fsb_pay = {
	clr_country_flag = ac_fsb_payment_succeeded
	if = {
		limit = {
			ac_fsb_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_enabled = no
			}
			set_country_flag = ac_fsb_payment_succeeded
		}
		else = {
			if = {
				limit = {
					ac_macro_ready = yes
				}
				ac_econ_pay_programme = yes
				if = {
					limit = {
						has_country_flag = ac_macro_transaction_succeeded
					}
					set_country_flag = ac_fsb_payment_succeeded
				}
			}
			else_if = {
				limit = {
					ac_econ_legacy_ready = yes
				}
				ac_econ_pay_programme = yes
				set_country_flag = ac_fsb_payment_succeeded
			}
		}
	}
}

	if = {
		limit = {
			ac_fsb_participant = yes any_owned_state = { always = yes } OR = { tag = FAT has_country_flag = ac_release_administration_registered } NOT = { has_country_flag = ac_fsb_initialized }
		}
		ac_pol_initialize = yes
		set_country_flag = ac_fsb_initialized
		set_variable = { ac_fsb_identity = THIS.id }
		set_variable = { ac_fsb_fiscal_applied = 0 }
		set_country_flag = ac_fsb_fiscal_dirty
		if = {
			limit = {
				date > 1991.1.7
			}

		}
		set_variable = { ac_fsb_warrants_open = 0 }
		set_variable = { ac_fsb_cases_open = 0 }
		set_variable = { ac_fsb_deliveries = 0 }
		set_variable = { ac_fsb_cases_resolved = 0 }
	}
}

ac_fsb_sync_commitments = {
	if = {
		limit = {
			ac_fsb_participant = yes check_variable = { ac_fsb_identity = THIS.id } has_country_flag = ac_fsb_fiscal_dirty ac_macro_ready = yes
		}
		set_temp_variable = { ac_macro_commitment_delta = ac_fsb_fiscal_applied }
		multiply_temp_variable = { ac_macro_commitment_delta = -1 }
		ac_macro_change_commitment = yes
		set_variable = { ac_fsb_fiscal_applied = 0 }
		if = {
			limit = {
				tag = FAT has_country_flag = ac_fsb_independent_audit_standing
			}
			set_temp_variable = { ac_fsb_commitment_piece = ac_macro_nominal_gdp }
			divide_temp_variable = { ac_fsb_commitment_piece = 2000 }
			add_to_variable = { ac_fsb_fiscal_applied = ac_fsb_commitment_piece }
		}
		if = {
			limit = {
				tag = SBK has_country_flag = ac_fsb_civil_charter_standing
			}
			set_temp_variable = { ac_fsb_commitment_piece = ac_macro_nominal_gdp }
			divide_temp_variable = { ac_fsb_commitment_piece = 2000 }
			add_to_variable = { ac_fsb_fiscal_applied = ac_fsb_commitment_piece }
		}
		if = {
			limit = {
				tag = SBK has_country_flag = ac_fsb_maintenance_account_standing
			}
			set_temp_variable = { ac_fsb_commitment_piece = ac_macro_nominal_gdp }
			divide_temp_variable = { ac_fsb_commitment_piece = 2000 }
			add_to_variable = { ac_fsb_fiscal_applied = ac_fsb_commitment_piece }
		}
		if = {
			limit = {
				tag = SBK has_country_flag = ac_fsb_service_account_standing
			}
			set_temp_variable = { ac_fsb_commitment_piece = ac_macro_nominal_gdp }
			divide_temp_variable = { ac_fsb_commitment_piece = 2000 }
			add_to_variable = { ac_fsb_fiscal_applied = ac_fsb_commitment_piece }
		}
		if = {
			limit = {
				tag = SBK has_country_flag = ac_fsb_security_remit_standing
			}
			set_temp_variable = { ac_fsb_commitment_piece = ac_macro_nominal_gdp }
			divide_temp_variable = { ac_fsb_commitment_piece = 2000 }
			add_to_variable = { ac_fsb_fiscal_applied = ac_fsb_commitment_piece }
		}
		set_temp_variable = { ac_macro_commitment_delta = ac_fsb_fiscal_applied }
		ac_macro_change_commitment = yes
		clr_country_flag = ac_fsb_fiscal_dirty
	}

ac_fsb_weekly = {
	ac_fsb_initialize = yes
	if = {
		limit = {
			ac_fsb_participant = yes has_country_flag = ac_fsb_initialized check_variable = { ac_fsb_identity = THIS.id }
		}
		if = {
			limit = {
				ac_fsb_active = no
			}
			ac_fsb_interrupt = yes
		}
		if = {
			limit = {
				ac_fsb_active = yes
			}
			ac_fsb_warrant_tick = yes
			ac_fsb_case_tick = yes
			ac_fsb_adviser_tick = yes
			ac_fsb_narrative_tick = yes
			ac_fsb_sync_commitments = yes
		}
	}
}
