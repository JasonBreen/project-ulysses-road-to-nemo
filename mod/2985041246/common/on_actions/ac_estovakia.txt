# Original provisional gameplay; source: tools/content_waves (The Estovakian Civil War). Native acceptance pending.
on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = EVA } EVA = { ac_ecw_initialize = yes } }
			if = { limit = { country_exists = EMM } EMM = { ac_ecw_initialize = yes } }
		}
	}
	on_weekly_EVA = {
		effect = {
			ac_ecw_weekly_eva = yes
		}
	}
	on_weekly_EMM = {
		effect = {
			ac_ecw_weekly_emm = yes
		}
	}
	on_annex = {
		effect = {
			FROM = { ac_ecw_interrupt = yes }
		}
	}
	on_capitulation_immediate = {
		effect = {
			ac_ecw_interrupt = yes
		}
	}
	on_subject_annexed = {
		effect = {
			ac_ecw_interrupt = yes
		}
	}
	on_war_relation_added = {
		effect = {
			ac_ecw_interrupt_service = yes
			FROM = { ac_ecw_interrupt_service = yes }
		}
	}
}
