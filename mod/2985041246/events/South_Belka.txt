add_namespace = south_belka

country_event = {
	id = south_belka.2
	title = south_belka.2.t
	desc = south_belka.2.d
	picture = GFX_report_event_belka_meme_read_write
	is_triggered_only = yes
	fire_only_once = yes
	option = {
		name = south_belka.2.a
		if = {
			limit = {
				NOT = {
					has_global_flag = ac_release_SBK_entry_begun
				}
				NOT = {
					has_global_flag = ac_release_SBK_adopted
				}
			}
			news_event = {
				id = ac_news_event.11
				days = 1
			}
			SBK = {
				inherit_technology = BLK
				transfer_state = 330
				transfer_state = 369
				transfer_state = 302
				transfer_state = 8
				transfer_state = 364
				country_event = {
					id = south_belka.1
					days = 10
				}
			}
			BLK = {
				if = {
					limit = {
						country_exists = SBK
						NOT = {
							has_global_flag = ac_release_SBK_transfer_done
						}
					}
					transfer_units_fraction = {
						target = SBK
						target_organization = 1
						source_organization = 1
						keep_unit_leaders_trigger = {
							always = yes
						}
						size = 0.25
						navy_ratio = 0
					}
					set_global_flag = ac_release_SBK_transfer_done
				}
				country_event = {
					id = belka_dlp_war.99
					days = 1
				}
			}
			set_global_flag = ac_release_SBK_entry_begun
			SBK = {
				ac_release_observe_country = yes
			}
		}
	}
	immediate = {
		ac_release_snapshot_world = yes
	}
	trigger = {
		NOT = {
			has_global_flag = ac_release_SBK_entry_begun
		}
		NOT = {
			has_global_flag = ac_release_SBK_adopted
		}
	}
}
