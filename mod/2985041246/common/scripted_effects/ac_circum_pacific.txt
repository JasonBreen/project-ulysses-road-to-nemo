ac_cpw_restore = {
	if = {
		limit = {
			ac_cpw_war_running = yes
		}
		if = {
			limit = {
				tag = YUK NOT = { has_idea = ac_cpw_yuk_war_footing }
			}
			add_ideas = ac_cpw_yuk_war_footing
		}
		if = {
			limit = {
				country_exists = OSE OSE = { NOT = { has_idea = ac_cpw_ose_war_footing } }
			}
			OSE = { add_ideas = ac_cpw_ose_war_footing }
		}
	}
}

ac_cpw_summit_tick = {
	if = {
		limit = {
			date > 2008.8.20 NOT = { has_global_flag = ac_cpw_summit_held } NOT = { has_global_flag = ac_cpw_summit_skipped }
		}
		if = {
			limit = {
				date < 2009.8.21 ac_cpw_principals = yes has_global_flag = ac_sp_arkbird_complete NOT = { has_war_with = OSE }
			}
			set_global_flag = ac_cpw_summit_held
			news_event = { id = ac_cpw.14 days = 1 }
		}
		else_if = {
			limit = {
				date > 2009.8.21
			}
			set_global_flag = ac_cpw_summit_skipped
		}
	}
}

ac_cpw_declaration_tick = {
	if = {
		limit = {
			has_global_flag = ac_cpw_calendar_started date > 2010.9.26 NOT = { has_global_flag = ac_cpw_started } NOT = { has_global_flag = ac_cpw_prevented } NOT = { has_global_flag = ac_cpw_declaration_withheld } NOT = { has_global_flag = ac_cpw_closed }
		}
		if = {
			limit = {
				ac_cpw_pair_at_war = yes
			}
			ac_cpw_observe_existing_war = yes
		}
		else_if = {
			limit = {
				ac_cpw_exchange_earned = yes
			}
			ac_cpw_prevent = yes
		}
		else_if = {
			limit = {
				ac_cpw_can_declare = yes
			}
			if = {
				limit = {
					NOT = { has_country_flag = ac_cpw_declaration_asked }
				}
				set_country_flag = ac_cpw_declaration_asked
				set_country_flag = ac_cpw_declaration_pending
				set_country_flag = { flag = ac_cpw_declaration_window days = 14 }
				country_event = { id = ac_cpw.5 days = 1 }
			}
			else_if = {
				limit = {
					has_country_flag = ac_cpw_declaration_pending NOT = { has_country_flag = ac_cpw_declaration_window }
				}
				ac_cpw_declare = yes
			}
		}
		else = {
			set_global_flag = ac_cpw_closed
			set_global_flag = ac_cpw_principal_unable
		}
	}
}
