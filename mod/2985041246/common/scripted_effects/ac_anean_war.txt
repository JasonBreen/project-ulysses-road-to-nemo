ac_eew_initialize = {
	if = {
		limit = {
			ac_eew_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_eew_current_service }
			}
			set_variable = { ac_eew_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_eew_elapsed }
			}
			set_variable = { ac_eew_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_eew_completed_services }
			}
			set_variable = { ac_eew_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_eew_last_delivered_days }
			}
			set_variable = { ac_eew_last_delivered_days = 0 }
		}
		set_country_flag = ac_eew_initialized
	}
}

ac_eew_pay_programme = {
	clr_country_flag = ac_eew_payment_authorized
	if = {
		limit = {
			ac_eew_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_eew_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_eew_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_eew_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_eew_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_eew_service_active has_variable = ac_eew_service_started
		}
		set_variable = { ac_eew_elapsed = date }
		subtract_from_variable = { ac_eew_elapsed = ac_eew_service_started }
		clamp_variable = { var = ac_eew_elapsed min = 0 max = 90 }
	}
}

	if = {
		limit = {
			tag = EMM ac_eew_active = yes OR = { has_country_flag = EMM_ew_evacuation_register_done has_country_flag = EMM_ew_administration_register_done has_country_flag = EMM_ew_reserve_register_done has_country_flag = EMM_ew_airforce_register_done has_country_flag = EMM_ew_liberation_register_done has_country_flag = EMM_ew_peace_register_done } NOT = { has_country_flag = ac_eew_message_50_sent }
		}
		set_country_flag = ac_eew_message_50_sent
		set_country_flag = ac_eew_message_50_pending
		set_country_flag = { flag = ac_eew_message_50_window days = 60 }
		country_event = { id = ac_eew.50 days = 1 }
	}

ac_eew_calendar_tick = {
	if = {
		limit = {
			date > 2015.4.19 NOT = { has_global_flag = ac_eew_calendar_started } NOT = { has_global_flag = ac_eew_closed }
		}
		if = {
			limit = {
				date < 2015.8.30 ac_eew_principals = yes ac_eew_sourced_route = yes
			}
			set_global_flag = ac_eew_calendar_started
			if = {
				limit = {
					has_global_flag = ac_anea_accord
				}
				set_global_flag = ac_eew_had_accord
			}
			if = {
				limit = {
					has_global_flag = ac_anea_bureau
				}
				set_global_flag = ac_eew_had_bureau
			}
			if = {
				limit = {
					has_global_flag = ac_anea_nec_participates
				}
				set_global_flag = ac_eew_nec_consulted
			}
		}
		else_if = {
			limit = {
				date < 2015.8.30 has_global_flag = ac_ecw_negotiated ac_eew_principals = yes
