ac_ecw_initialize = {
	if = {
		limit = {
			ac_ecw_eligible = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ac_ecw_current_service }
			}
			set_variable = { ac_ecw_current_service = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_ecw_elapsed }
			}
			set_variable = { ac_ecw_elapsed = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_ecw_completed_services }
			}
			set_variable = { ac_ecw_completed_services = 0 }
		}
		if = {
			limit = {
				NOT = { has_variable = ac_ecw_last_delivered_days }
			}
			set_variable = { ac_ecw_last_delivered_days = 0 }
		}
		set_country_flag = ac_ecw_initialized
	}
}

ac_ecw_pay_programme = {
	clr_country_flag = ac_ecw_payment_authorized
	if = {
		limit = {
			ac_ecw_can_fund = yes
		}
		if = {
			limit = {
				ac_econ_ready = yes
			}
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_ecw_payment_authorized
			}
		}
		else = {
			set_country_flag = ac_ecw_payment_authorized
		}
		if = {
			limit = {
				has_country_flag = ac_ecw_payment_authorized
			}
			add_political_power = -35
			set_temp_variable = { ac_pol_capacity_delta = -4 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_ecw_refresh_progress = {
	if = {
		limit = {
			has_country_flag = ac_ecw_service_active has_variable = ac_ecw_service_started
		}
		set_variable = { ac_ecw_elapsed = date }
		subtract_from_variable = { ac_ecw_elapsed = ac_ecw_service_started }
		clamp_variable = { var = ac_ecw_elapsed min = 0 max = 90 }
	}
}

ac_ecw_update_tier = {
	if = {
		limit = {
			ac_ecw_running = yes check_variable = { ac_ecw_luf_authority > 59.99 }
		}
		if = {
			limit = {
				NOT = { has_idea = ac_ecw_luf_holds_capital }
			}
			remove_ideas = ac_ecw_luf_holds_capital
			remove_ideas = ac_ecw_divided_country
			remove_ideas = ac_ecw_luf_in_retreat
			add_ideas = ac_ecw_luf_holds_capital
		}
	}
	else_if = {
		limit = {
			ac_ecw_running = yes check_variable = { ac_ecw_luf_authority > 29.99 }
		}
		if = {
			limit = {
				NOT = { has_idea = ac_ecw_divided_country }
			}
			remove_ideas = ac_ecw_luf_holds_capital
			remove_ideas = ac_ecw_divided_country
			remove_ideas = ac_ecw_luf_in_retreat
			add_ideas = ac_ecw_divided_country
		}
	}
	else_if = {
		limit = {
			ac_ecw_running = yes
		}
		if = {
			limit = {
				NOT = { has_idea = ac_ecw_luf_in_retreat }
			}
			remove_ideas = ac_ecw_luf_holds_capital
			remove_ideas = ac_ecw_divided_country
			remove_ideas = ac_ecw_luf_in_retreat
			add_ideas = ac_ecw_luf_in_retreat
		}
	}
	else = {
		remove_ideas = ac_ecw_luf_holds_capital
		remove_ideas = ac_ecw_divided_country
		remove_ideas = ac_ecw_luf_in_retreat
	}
}
