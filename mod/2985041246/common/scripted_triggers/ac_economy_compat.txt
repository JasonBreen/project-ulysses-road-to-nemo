}

	OR = { ac_econ_legacy_can_order = yes ac_macro_can_fund_replacements = yes }
}

	OR = { ac_econ_legacy_can_produce = yes ac_macro_can_produce = yes }
}

	OR = { ac_econ_legacy_can_repay = yes ac_macro_can_repay = yes }
}

}

	OR = { ac_econ_legacy_ready = yes ac_macro_ready = yes }
}
ac_econ_can_pay_programme = {
	OR = { AND = { ac_macro_ready = yes check_variable = { ac_macro_arrears = 0 } check_variable = { ac_macro_cash >= 0.002 } } AND = { ac_econ_legacy_ready = yes check_variable = { ac_econ_arrears = 0 } check_variable = { ac_econ_treasury >= 2 } } }
}
