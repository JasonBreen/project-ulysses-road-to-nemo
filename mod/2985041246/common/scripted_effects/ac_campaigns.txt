ac_camp_pay = {
	clr_country_flag = ac_camp_paid
	if = {
		limit = {
			ac_camp_can_pay = yes
		}
		if = {
			limit = {
				ac_econ_enabled = no
			}
			set_country_flag = ac_camp_paid
		}
		else = {
			ac_econ_pay_programme = yes
			if = {
				limit = {
					OR = { ac_macro_ready = no has_country_flag = ac_macro_transaction_succeeded }
				}
				set_country_flag = ac_camp_paid
			}
		}
		if = {
			limit = {
				has_country_flag = ac_camp_paid
			}
			add_political_power = -25
			set_temp_variable = { ac_pol_capacity_delta = -2 }
			ac_pol_change_capacity = yes
		}
	}
}

ac_camp_initialize = {
	if = {
		limit = {
			ac_camp_eligible = yes NOT = { has_country_flag = ac_camp_initialized }
		}
		set_country_flag = ac_camp_initialized
		set_variable = { ac_camp_identity = THIS.id }
		if = {
			limit = {
				OR = { AND = { tag = OSE date > 2010.11.28 } AND = { tag = ERU date > 2005.4.1 } }
			}
			set_country_flag = ac_camp_late_character_history
		}
	}
}

ac_camp_relief_start = {
	if = {
		limit = {
			ac_camp_relief_can_start = yes
		}
		FCU = { ac_camp_pay = yes }
		if = {
			limit = {
				FCU = { has_country_flag = ac_camp_paid }
			}
			set_country_flag = ac_camp_busy
			set_country_flag = ac_camp_relief_active
			set_country_flag = { flag = ac_camp_relief_waiting days = 7 }
			set_country_flag = { flag = ac_camp_relief_deadline days = 21 }
		}
	}
}

ac_camp_relief_finish = {
	if = {
		limit = {
			has_country_flag = ac_camp_relief_active NOT = { has_country_flag = ac_camp_relief_waiting } ac_camp_relief_can_finish = yes
		}
		clr_country_flag = ac_camp_relief_active
		clr_country_flag = ac_camp_busy
		set_global_flag = ac_camp_relief_success
		add_timed_idea = { idea = ac_camp_operation_return days = 90 }
	}
}

ac_camp_relief_tick = {
	if = {
		limit = {
			has_country_flag = ac_camp_relief_active
		}
		if = {
			limit = {
				ac_camp_relief_context = no
			}
			clr_country_flag = ac_camp_relief_active
			clr_country_flag = ac_camp_busy
			set_global_flag = ac_camp_relief_interrupted
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = ac_camp_relief_waiting } ac_camp_relief_can_finish = yes
			}
			ac_camp_relief_finish = yes
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = ac_camp_relief_deadline } has_global_flag = ac_camp_clinic_civil_permission NOT = { has_global_flag = ac_camp_clinic_military_permission }
			}
			clr_country_flag = ac_camp_relief_active
			clr_country_flag = ac_camp_busy
			set_global_flag = ac_camp_relief_limited
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = ac_camp_relief_deadline }
			}
			ac_camp_relief_fail = yes
		}
	}
}

ac_camp_character_tick = {
	if = {
		limit = {
			tag = ERU NOT = { has_country_flag = ac_camp_late_character_history } has_global_flag = ac_camp_battery_success NOT = { has_global_flag = ac_camp_yellow_four_settled }
		}
		set_global_flag = ac_camp_yellow_four_settled
		if = {
			limit = {
				has_global_flag = ac_camp_yellow_four_airworthy
			}
			set_global_flag = ac_camp_yellow_four_preserved
		}
		else = {
			set_global_flag = ac_camp_yellow_four_lost
		}
	}
	if = {
		limit = {
			tag = ERU NOT = { has_country_flag = ac_camp_late_character_history } has_global_flag = ac_ucw_outcome_isaf_victory NOT = { has_global_flag = ac_camp_yellow_thirteen_settled }
		}
		set_global_flag = ac_camp_yellow_thirteen_settled
		if = {
			limit = {
				NOT = { has_global_flag = ac_camp_yellow_thirteen_preserved }
			}
			set_global_flag = ac_camp_yellow_thirteen_lost
		}
	}
	if = {
		limit = {
			tag = OSE NOT = { has_country_flag = ac_camp_late_character_history } ac_cpw_course_open = yes date > 2010.11.28 NOT = { has_global_flag = ac_camp_chopper_settled }
		}
		set_global_flag = ac_camp_chopper_settled
		if = {
			limit = {
				has_global_flag = ac_camp_rescue_cover_success
			}
			set_global_flag = ac_camp_chopper_preserved
		}
		else = {
			set_global_flag = ac_camp_chopper_lost
		}
	}
	if = {
		limit = {
			tag = BLK has_global_flag = ac_camp_reckoning_success OR = { has_global_flag = ac_cpw_ended has_global_flag = ac_cpw_prevented has_global_flag = ac_cpw_declaration_withheld has_global_flag = ac_cpw_late_start_skipped has_global_flag = ac_cpw_principal_unable } NOT = { has_global_flag = ac_camp_network_settled }
		}
		set_global_flag = ac_camp_network_settled
		if = {
			limit = {
				has_global_flag = ac_camp_seizure_success NOT = { has_global_flag = ac_camp_evidence_success } NOT = { has_global_flag = ac_camp_witness_success }
			}
			set_global_flag = ac_camp_network_success_ending
		}
		else = {
			set_global_flag = ac_camp_network_exposed_ending
		}
	}
}
