# Authored gameplay; source: tools/usean_partners/content.py and build.py.
on_actions = {
	on_startup = {
		effect = {
			SNS = { ac_up_initialize = yes }
			AMB = { ac_up_initialize = yes }
			UGE = { ac_up_initialize = yes }
			DEL = { ac_up_initialize = yes }
		}
	}
	on_weekly_SNS = {
		effect = {
			ac_up_weekly = yes
		}
	}

	on_weekly_AMB = {
		effect = {
			ac_up_weekly = yes
		}
	}

	on_weekly_UGE = {
		effect = {
			ac_up_weekly = yes
		}
	}

	on_weekly_DEL = {
		effect = {
			ac_up_weekly = yes
		}
	}
	on_weekly_FCU = {
		effect = {
			ac_up_fcu_cleanup = yes
		}
	}
	on_annex = {
		effect = {
			FROM = {
				ac_up_interrupt_services = yes
				ac_up_fcu_interrupt = yes
			}
		}
	}
	on_capitulation_immediate = {
		effect = {
			ac_up_interrupt_services = yes
			ac_up_fcu_interrupt = yes
		}
	}
	on_subject_annexed = {
		effect = {
			ac_up_interrupt_services = yes
			ac_up_fcu_interrupt = yes
		}
	}
	on_puppet = {
		effect = {
			ac_up_status_diplomacy_interrupt = yes
		}
	}
	on_subject_free = {
		effect = {
			ac_up_status_diplomacy_interrupt = yes
		}
	}
	on_subject_autonomy_level_change = {
		effect = {
			if = {
				limit = { is_subject = yes }
				ac_up_status_diplomacy_interrupt = yes
			}
		}
	}
	on_war_relation_added = {
		effect = {
			ac_up_diplomacy_tick = yes
			ac_up_fcu_cleanup = yes
			FROM = {
				ac_up_diplomacy_tick = yes
				ac_up_fcu_cleanup = yes
			}
		}
	}
}
