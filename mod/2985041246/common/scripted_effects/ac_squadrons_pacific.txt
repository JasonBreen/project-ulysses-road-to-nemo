ac_sqp_initialize = {
	if = {
		limit = {
			tag = OSE original_tag = OSE country_exists = OSE ac_sqp_enabled = yes NOT = { has_country_flag = ac_sqp_initialized }
		}
		set_country_flag = ac_sqp_initialized
		set_variable = { ac_sqp_identity = THIS.id }
		set_variable = { ac_sqp_chapter = 0 }
		set_variable = { ac_sqp_pending_chapter = 0 }
		set_variable = { ac_sqp_ending_kind = -1 }
		set_variable = { ac_sqp_cohesion = 50 }
		set_variable = { ac_sqp_credibility = 50 }
		if = {
			limit = {
				date > 2020.12.31
			}
			set_country_flag = ac_sqp_closed
			set_country_flag = ac_sqp_late_archive
		}
		else_if = {
			limit = {
				date > 1991.1.31
			}
			set_variable = { ac_sqp_chapter = 1 }
			set_country_flag = ac_sqp_ose_missed_1
		}
	}
}

ac_sqp_ose_finish = {
	if = {
		limit = {
			tag = OSE original_tag = OSE country_exists = OSE ac_sqp_active = yes check_variable = { ac_sqp_chapter = 20 } NOT = { has_country_flag = ac_sqp_ending_pending }
		}
		set_country_flag = { flag = ac_sqp_ending_pending days = 30 }
		if = {
			limit = {
				NOT = { check_variable = { ac_sqp_cohesion < 65 } } NOT = { check_variable = { ac_sqp_credibility < 65 } }
			}
			set_variable = { ac_sqp_ending_kind = 0 }
			country_event = { id = ac_sqp.790 days = 1 }
		}
		else_if = {
			limit = {
				check_variable = { ac_sqp_cohesion < 45 }
			}
			set_variable = { ac_sqp_ending_kind = 2 }
			country_event = { id = ac_sqp.792 days = 1 }
		}
		else = {
			set_variable = { ac_sqp_ending_kind = 1 }
			country_event = { id = ac_sqp.791 days = 1 }
		}
	}
}
ac_sqp_tick = {
	ac_sqp_initialize = yes
	if = {
		limit = {
			tag = OSE original_tag = OSE country_exists = OSE ac_sqp_active = yes
		}
		ac_sqp_ose_repair_pending = yes
	}
	if = {
		limit = {
			tag = OSE original_tag = OSE country_exists = OSE ac_sqp_active = yes NOT = { has_country_flag = ac_sqp_paused }
		}
		ac_sqp_ose_dispatch = yes
	}
	if = {
		limit = {
			tag = OSE original_tag = OSE country_exists = OSE NOT = { has_country_flag = ac_sqp_paused }
		}
		ac_sqp_ose_finish = yes
	}
}
