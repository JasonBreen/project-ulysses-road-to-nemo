ac_sq_initialize = {
	if = {
		limit = {
			ac_sq_enabled = yes ac_sq_host = yes NOT = { has_country_flag = ac_sq_initialized }
		}
		set_country_flag = ac_sq_initialized
		set_variable = { ac_sq_chapter = 0 }
		set_variable = { ac_sq_pending_chapter = 0 }
		set_variable = { ac_sq_ending_kind = -1 }
		set_variable = { ac_sq_cohesion = 50 }
		set_variable = { ac_sq_credibility = 50 }
		if = {
			limit = {
				date > 1991.1.31 NOT = { date > 2005.12.31 }
			}
			if = {
				limit = {
					tag = BLK original_tag = BLK country_exists = BLK
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_blk_missed_1
			}
			if = {
				limit = {
					tag = UST original_tag = UST country_exists = UST
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_ust_missed_1
			}
			if = {
				limit = {
					tag = ERU original_tag = ERU country_exists = ERU
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_eru_missed_1
			}
			if = {
				limit = {
					tag = SNS original_tag = SNS country_exists = SNS
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_sns_missed_1
			}
			if = {
				limit = {
					tag = EMM original_tag = EMM country_exists = EMM
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_emm_missed_1
			}
			if = {
				limit = {
					tag = EVA original_tag = EVA country_exists = EVA
				}
				set_variable = { ac_sq_chapter = 1 }
				set_country_flag = ac_sq_eva_missed_1

ac_sq_eva_finish = {
	if = {
		limit = {
			tag = EVA original_tag = EVA country_exists = EVA ac_sq_active = yes check_variable = { ac_sq_chapter = 20 } NOT = { has_country_flag = ac_sq_ending_pending }
		}
		set_country_flag = { flag = ac_sq_ending_pending days = 30 }
		if = {
			limit = {
				NOT = { check_variable = { ac_sq_cohesion < 65 } } NOT = { check_variable = { ac_sq_credibility < 65 } }
			}
			set_variable = { ac_sq_ending_kind = 0 }
			country_event = { id = ac_sq.690 days = 1 }
		}
		else_if = {
			limit = {
				check_variable = { ac_sq_cohesion < 45 }
			}
			set_variable = { ac_sq_ending_kind = 2 }
			country_event = { id = ac_sq.692 days = 1 }
		}
		else = {
			set_variable = { ac_sq_ending_kind = 1 }
			country_event = { id = ac_sq.691 days = 1 }
		}
	}
}
