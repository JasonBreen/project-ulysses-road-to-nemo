# Project Ulysses superweapon completion effects (country scope).
# Shared by the special project outputs (common/special_projects/projects/
# ac_strangereal_projects.txt) and the no-DLC fallback decisions
# (common/decisions/ac_superweapons.txt) so both paths grant the same reward.
# Each effect is idempotent. Reward coefficients are provisional gameplay.
# Only the six ac_sp_*_complete global flags are set here.

# Chronicle 1994 / 12.23; strengthens the focus-built Excalibur (mod provenance).
ac_sp_excalibur_complete_effect = {
	# The target may be destroyed or seized while a programme runs.
	if = {
		limit = {
			NOT = { has_global_flag = excalibur_destroyed }
		}
		set_global_flag = ac_sp_excalibur_complete
		if = {
			limit = { NOT = { has_idea = ac_sp_excalibur_hardened } NOT = { has_idea = ac_sp_rhongomyniad_tower } }
			add_ideas = ac_sp_excalibur_hardened
		}
	}
}

# Chronicle 1995 / 05.17 and 12.31. The inherited nuclear chain
# (belka_dlp_war.99/.100, belka_dropped_nukes) is not changed; later content may
# read ac_sp_v2_complete.
ac_sp_v2_complete_effect = {
	# The V2 launch facilities were attacked on 1995 / 12.31.
	if = {
		limit = {
			date < 1996.1.1
		}
		set_global_flag = ac_sp_v2_complete
		if = {
			limit = { NOT = { has_idea = ac_sp_v2_deterrent } }
			add_ideas = ac_sp_v2_deterrent
		}
	}
}

# Chronicle 1998 / 06.19. Feeds The Peace We Inherit only when its variables exist.
ac_sp_stonehenge_complete_effect = {
	# The target may be destroyed or seized while a programme runs.
	if = {
		limit = {
			NOT = { has_global_flag = ac_ucw_stonehenge_seized }
			NOT = { has_global_flag = ac_ucw_stonehenge_destroyed }
		}
		set_global_flag = ac_sp_stonehenge_complete
		if = {
			limit = { NOT = { has_idea = ac_sp_stonehenge_battery } }
			add_ideas = ac_sp_stonehenge_battery
			if = {
				limit = { has_variable = ac_pwi_preparedness NOT = { has_country_flag = ac_pwi_crisis_resolved } }
				add_to_variable = { ac_pwi_preparedness = 15 }
				ac_pwi_bound_pressures = yes
			}
			else_if = {
				limit = { has_variable = ac_pwi_relief_capacity }
				add_to_variable = { ac_pwi_relief_capacity = 10 }
			}
		}
	}
}

# Chronicle 1999 / 10.08. Debris clearing only; no weapon. The Yuktobanian
# co-operation bonus applies only when Track B has set ac_yuk_arkbird_partner.
ac_sp_arkbird_complete_effect = {
	set_global_flag = ac_sp_arkbird_complete
	if = {
		limit = { NOT = { has_idea = ac_sp_arkbird_debris_clearing } }
		add_ideas = ac_sp_arkbird_debris_clearing
		if = {
			limit = {
				country_exists = YUK
				YUK = { has_country_flag = ac_yuk_arkbird_partner }
			}
			add_ideas = ac_sp_arkbird_joint_operations
			YUK = {
				if = {
					limit = { NOT = { has_idea = ac_sp_arkbird_joint_operations } }
					add_ideas = ac_sp_arkbird_joint_operations
				}
			}
		}
	}
}

# Provisional gameplay: gives substance to BLK_project_rhongomyniad /
# belka_dlp_won.9 (mod provenance). Not an official ACE COMBAT weapon.
ac_sp_rhongomyniad_complete_effect = {
	# The target may be destroyed or seized while a programme runs.
	if = {
		limit = {
			NOT = { has_global_flag = excalibur_destroyed }
		}
		set_global_flag = ac_sp_rhongomyniad_complete
		if = {
			limit = { NOT = { has_idea = ac_sp_rhongomyniad_tower } }
			if = {
				limit = { has_idea = ac_sp_excalibur_hardened }
				remove_ideas = ac_sp_excalibur_hardened
			}
			add_ideas = ac_sp_rhongomyniad_tower
		}
	}
}
