# Follows the existing ac_postwar_local_recovery decision and the ac_pwi.76 choice.
shared_focus = {
	id = REC_local_recovery_programme
	icon = GFX_ac_briefing_ose_industrial_buildup
	cost = 5
	ai_will_do = { factor = 90 }
	prerequisite = { focus = REC_consolidate_the_republic }
	x = 4
	y = 1
	available = {
		has_war = no
		custom_trigger_tooltip = {
			tooltip = REC_postwar_handoff_tt
			has_global_flag = ac_postwar_contract_started
		}
		# Outcomes that never assign a settlement decision pass this check.
		custom_trigger_tooltip = {
			tooltip = REC_local_settlement_resolved_tt
			NOT = { has_country_flag = ac_postwar_settlement_pending }
		}
		# REC is always an ac_pwi participant, so ac_pwi.76 always arrives once it is at peace.
		custom_trigger_tooltip = {
			tooltip = REC_local_briefing_answered_tt
			OR = {
				has_country_flag = ac_pwi_choice_76_0
				has_country_flag = ac_pwi_choice_76_1
			}
		}
	}
	completion_reward = {
		if = {
			limit = { ac_pwi_active = yes }
			add_to_variable = { ac_pwi_recovery = 5 }
			if = {
				limit = { has_country_flag = ac_pwi_choice_76_0 }
				add_to_variable = { ac_pwi_recovery = 3 }
			}
			else_if = {
				limit = { has_country_flag = ac_pwi_choice_76_1 }
				add_to_variable = { ac_pwi_confidence = 3 }
			}
			ac_pwi_bound_pressures = yes
		}
		random_owned_controlled_state = {
			add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
		}
		if = {
			limit = { ac_econ_ready = yes }
			ac_econ_set_reconstruction = yes
		}
	}
	search_filters = { }
}
